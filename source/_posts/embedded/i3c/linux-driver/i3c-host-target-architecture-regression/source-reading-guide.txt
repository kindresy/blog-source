 871      int ret;
 872      if (t != &target) {
 873          return -EINVAL;
 874      }
 875      if (t->state == I3C_TARGET_CLOSED) {
 876          return 0;
 877      }
 878      /* 即使Host尚未来、FIFO为空，也不能关闭仍有软件收发请求的会话。
 879       * PIO可先cancel解除等待；IBI不能写0撤销，必须完成或平台复位。 */
 880      if (t->ibi_owned || t->reset_owned || t->dma_owned || t->mrl.active ||
 881          i3c_target_rx_active(t) || i3c_target_tx_active(t)) {
 882          return -EBUSY;
 883      }
 884      ret = take_snapshot(t, &s);
 885      if (ret) {
 886          return ret; /* 不在读回失败时猜测已停止。 */
 887      }
 888      if (!quiet(&s)) {
 889          return -EBUSY;
 890      }
 891      if (t->state != I3C_TARGET_CLOSING) {
 892          /* ack尚为0可能只是ENABLE请求未跨到SCL域，不能据此断钟。
 893           * 先让Host提供SCL并观察到ack=1，再请求关闭，才能判关闭ack。
 894           * 本M1阶段没有reset/强制abort兜底，也不会清任何队列。 */
 895          if (!(s.ctrl & ENABLE)) {
 896              return -EAGAIN;
 897          }
 898          if (!t->enable_attempted) {
 899              return -EBUSY;
 900          }
 901          t->ctrl_shadow &= ~ENABLE;
 902          /* MMIO报错不保证写请求未生效。先记录已尝试关闭，保留所有权，
 903           * 后续close仍可接受迟到的ack=0；ack不降时保持pending，由Host
 904           * 提供SCL或执行板级恢复，不能因软件报错直接断钟。 */
 905          t->state = I3C_TARGET_CLOSING;
 906          ret = write_reg(t, CTRL, t->ctrl_shadow);
 907          if (ret) {
 908              t->last_error = ret;
 909              return ret;
 910          }
 911          I3C_LOG_DEBUG("I3C_RTL_CLOSE_NEED_HOST_CLOCK: disable_requested=1 stopped=0\n");
 912          return -EINPROGRESS;
 913      }

 915          return -EINPROGRESS;
 916      }

 919      if (ret) {

 921      }
